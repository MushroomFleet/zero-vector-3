What You'll Build

Automated Risk Detection

Implement comprehensive risk detection that identifies content requiring human oversight:

// Risk detection engine
export class RiskDetectionEngine {
    private riskRules: RiskRule[];
    private contentAnalyzer: ContentAnalyzer;
    private contextAnalyzer: ContextAnalyzer;

    constructor(config: RiskDetectionConfig) {
        this.riskRules = this.loadRiskRules(config.rulesPath);
        this.contentAnalyzer = new ContentAnalyzer(config.analysis);
        this.contextAnalyzer = new ContextAnalyzer(config.context);
    }

    async assessRisk(
        workflowOutput: WorkflowOutput,
        context: WorkflowContext
    ): Promise {
        // Analyze content for risk indicators
        const contentRisk = await this.analyzeContentRisk(workflowOutput);
        
        // Analyze context for approval triggers
        const contextRisk = await this.analyzeContextRisk(context);
        
        // Apply business rules
        const ruleRisk = await this.applyRiskRules(workflowOutput, context);
        
        // Calculate overall risk score
        const overallRisk = this.calculateOverallRisk(
            contentRisk,
            contextRisk,
            ruleRisk
        );

        return {
            overallScore: overallRisk.score,
            riskLevel: overallRisk.level,
            triggers: overallRisk.triggers,
            requiredApprovals: this.determineRequiredApprovals(overallRisk),
            recommendations: this.generateRecommendations(overallRisk)
        };
    }

    private async analyzeContentRisk(
        output: WorkflowOutput
    ): Promise {
        const risks = [];

        // Financial content analysis
        const financialRisk = await this.contentAnalyzer.detectFinancialAdvice(output.content);
        if (financialRisk.detected) {
            risks.push({
                type: 'financial_advice',
                score: financialRisk.confidence,
                details: financialRisk.specificConcerns
            });
        }

        // Medical content analysis
        const medicalRisk = await this.contentAnalyzer.detectMedicalAdvice(output.content);
        if (medicalRisk.detected) {
            risks.push({
                type: 'medical_advice',
                score: medicalRisk.confidence,
                details: medicalRisk.specificConcerns
            });
        }

        // Legal content analysis
        const legalRisk = await this.contentAnalyzer.detectLegalAdvice(output.content);
        if (legalRisk.detected) {
            risks.push({
                type: 'legal_advice',
                score: legalRisk.confidence,
                details: legalRisk.specificConcerns
            });
        }

        // Sensitive information detection
        const sensitiveRisk = await this.contentAnalyzer.detectSensitiveInfo(output.content);
        if (sensitiveRisk.detected) {
            risks.push({
                type: 'sensitive_information',
                score: sensitiveRisk.confidence,
                details: sensitiveRisk.specificConcerns
            });
        }

        return {
            risks: risks,
            maxScore: Math.max(...risks.map(r => r.score), 0),
            categories: risks.map(r => r.type)
        };
    }

    private async analyzeContextRisk(
        context: WorkflowContext
    ): Promise {
        const contextFactors = [];

        // User role and permissions
        if (context.user?.role === 'external' || !context.user?.verified) {
            contextFactors.push({
                type: 'user_verification',
                score: 0.8,
                reason: 'Unverified or external user requesting sensitive operation'
            });
        }

        // Request magnitude
        if (context.requestValue && context.requestValue > context.thresholds?.high) {
            contextFactors.push({
                type: 'high_value_request',
                score: 0.9,
                reason: `Request value ${context.requestValue} exceeds threshold`
            });
        }

        // Unusual patterns
        const patternRisk = await this.contextAnalyzer.detectUnusualPatterns(context);
        if (patternRisk.score > 0.6) {
            contextFactors.push({
                type: 'unusual_pattern',
                score: patternRisk.score,
                reason: patternRisk.description
            });
        }

        return {
            factors: contextFactors,
            maxScore: Math.max(...contextFactors.map(f => f.score), 0),
            combinedScore: this.calculateCombinedContextScore(contextFactors)
        };
    }

    private async applyRiskRules(
        output: WorkflowOutput,
        context: WorkflowContext
    ): Promise {
        const applicableRules = this.riskRules.filter(rule => 
            rule.condition(output, context)
        );

        const ruleResults = applicableRules.map(rule => ({
            ruleId: rule.id,
            triggered: true,
            score: rule.riskScore,
            action: rule.action,
            reason: rule.reason
        }));

        return {
            triggeredRules: ruleResults,
            highestScore: Math.max(...ruleResults.map(r => r.score), 0),
            requiredActions: ruleResults.map(r => r.action)
        };
    }
}

Approval Engine

Core Approval Engine Implementation

// Core approval engine
export class ApprovalEngine {
    private approvalQueue: ApprovalQueue;
    private reviewerManager: ReviewerManager;
    private policyEngine: PolicyEngine;
    private notificationService: NotificationService;

    constructor(config: ApprovalEngineConfig) {
        this.approvalQueue = new ApprovalQueue(config.queue);
        this.reviewerManager = new ReviewerManager(config.reviewers);
        this.policyEngine = new PolicyEngine(config.policies);
        this.notificationService = new NotificationService(config.notifications);
    }

    async requestApproval(
        request: ApprovalRequest
    ): Promise {
        // Validate approval request
        await this.validateApprovalRequest(request);

        // Determine approval requirements
        const requirements = await this.determineApprovalRequirements(request);

        // Create approval session
        const session = await this.createApprovalSession(request, requirements);

        // Assign reviewers
        await this.assignReviewers(session);

        // Queue for review
        await this.approvalQueue.enqueue(session);

        // Send notifications
        await this.notifyReviewers(session);

        return {
            sessionId: session.id,
            status: 'pending_approval',
            estimatedReviewTime: session.estimatedReviewTime,
            assignedReviewers: session.assignedReviewers.map(r => r.id),
            approvalRequirements: requirements
        };
    }

    async processApprovalDecision(
        sessionId: string,
        decision: ApprovalDecision
    ): Promise {
        const session = await this.approvalQueue.getSession(sessionId);
        
        if (!session) {
            throw new Error(`Approval session ${sessionId} not found`);
        }

        // Validate decision authority
        await this.validateDecisionAuthority(session, decision);

        // Apply decision to session
        const updatedSession = await this.applyDecision(session, decision);

        // Check if all required approvals are complete
        const completionStatus = await this.checkCompletionStatus(updatedSession);

        if (completionStatus.isComplete) {
            // Process final approval result
            return await this.processFinalApproval(updatedSession, completionStatus);
        } else {
            // Continue approval process
            return await this.continueApprovalProcess(updatedSession, completionStatus);
        }
    }

    private async determineApprovalRequirements(
        request: ApprovalRequest
    ): Promise {
        const baseRequirements = await this.policyEngine.getBaseRequirements(
            request.riskAssessment
        );

        const domainRequirements = await this.policyEngine.getDomainRequirements(
            request.domain,
            request.context
        );

        const escalationRequirements = await this.policyEngine.getEscalationRequirements(
            request.riskAssessment.overallScore
        );

        return {
            approvalLevels: this.mergeApprovalLevels([
                baseRequirements.levels,
                domainRequirements.levels,
                escalationRequirements.levels
            ]),
            reviewerCriteria: this.mergeReviewerCriteria([
                baseRequirements.reviewerCriteria,
                domainRequirements.reviewerCriteria,
                escalationRequirements.reviewerCriteria
            ]),
            timeoutPolicy: this.selectTimeoutPolicy([
                baseRequirements.timeout,
                domainRequirements.timeout,
                escalationRequirements.timeout
            ]),
            escalationTriggers: this.mergeEscalationTriggers([
                baseRequirements.escalationTriggers,
                domainRequirements.escalationTriggers,
                escalationRequirements.escalationTriggers
            ])
        };
    }

    private async assignReviewers(
        session: ApprovalSession
    ): Promise {
        const requirements = session.requirements;
        const assignedReviewers = [];

        for (const level of requirements.approvalLevels) {
            const availableReviewers = await this.reviewerManager.findEligibleReviewers(
                level.criteria
            );

            if (availableReviewers.length === 0) {
                throw new Error(`No eligible reviewers found for approval level: ${level.name}`);
            }

            // Select optimal reviewer based on availability, expertise, and workload
            const selectedReviewer = await this.selectOptimalReviewer(
                availableReviewers,
                level,
                session
            );

            assignedReviewers.push({
                level: level.name,
                reviewer: selectedReviewer,
                status: 'assigned',
                assignedAt: new Date(),
                dueBy: new Date(Date.now() + level.timeoutMs)
            });
        }

        session.assignedReviewers = assignedReviewers;
    }

    private async selectOptimalReviewer(
        availableReviewers: Reviewer[],
        level: ApprovalLevel,
        session: ApprovalSession
    ): Promise {
        // Score reviewers based on multiple criteria
        const scoredReviewers = await Promise.all(
            availableReviewers.map(async reviewer => ({
                reviewer,
                score: await this.scoreReviewer(reviewer, level, session)
            }))
        );

        // Sort by score (highest first)
        scoredReviewers.sort((a, b) => b.score - a.score);

        return scoredReviewers[0].reviewer;
    }

    private async scoreReviewer(
        reviewer: Reviewer,
        level: ApprovalLevel,
        session: ApprovalSession
    ): Promise {
        let score = 0;

        // Expertise match
        const expertiseScore = this.calculateExpertiseScore(
            reviewer.expertise,
            level.criteria.requiredExpertise
        );
        score += expertiseScore * 0.4;

        // Availability
        const availability = await this.reviewerManager.getAvailability(reviewer.id);
        score += availability.score * 0.3;

        // Current workload
        const workload = await this.reviewerManager.getCurrentWorkload(reviewer.id);
        score += (1 - workload.normalized) * 0.2;

        // Historical performance
        const performance = await this.reviewerManager.getPerformanceMetrics(reviewer.id);
        score += performance.averageScore * 0.1;

        return score;
    }
}

Approval Session Management

// Approval session management
export class ApprovalSession {
    public id: string;
    public request: ApprovalRequest;
    public requirements: ApprovalRequirements;
    public assignedReviewers: AssignedReviewer[];
    public decisions: ApprovalDecision[];
    public status: ApprovalStatus;
    public createdAt: Date;
    public dueBy: Date;
    public estimatedReviewTime: number;
    public escalationHistory: EscalationEvent[];
    public modifications: ApprovalModification[];

    constructor(request: ApprovalRequest, requirements: ApprovalRequirements) {
        this.id = this.generateSessionId();
        this.request = request;
        this.requirements = requirements;
        this.assignedReviewers = [];
        this.decisions = [];
        this.status = 'pending_assignment';
        this.createdAt = new Date();
        this.dueBy = this.calculateDueDate(requirements);
        this.estimatedReviewTime = this.estimateReviewTime(requirements);
        this.escalationHistory = [];
        this.modifications = [];
    }

    addDecision(decision: ApprovalDecision): void {
        this.decisions.push({
            ...decision,
            timestamp: new Date()
        });

        this.updateStatus();
    }

    addModification(modification: ApprovalModification): void {
        this.modifications.push({
            ...modification,
            timestamp: new Date()
        });
    }

    escalate(reason: string, newAssignee: string): void {
        this.escalationHistory.push({
            reason,
            escalatedTo: newAssignee,
            escalatedAt: new Date(),
            escalatedBy: 'system'
        });

        this.status = 'escalated';
    }

    isComplete(): boolean {
        const requiredLevels = this.requirements.approvalLevels.map(l => l.name);
        const approvedLevels = this.decisions
            .filter(d => d.decision === 'approved')
            .map(d => d.level);

        return requiredLevels.every(level => approvedLevels.includes(level));
    }

    isRejected(): boolean {
        return this.decisions.some(d => d.decision === 'rejected');
    }

    isExpired(): boolean {
        return new Date() > this.dueBy;
    }

    getOverallDecision(): 'approved' | 'rejected' | 'pending' | 'expired' {
        if (this.isExpired()) return 'expired';
        if (this.isRejected()) return 'rejected';
        if (this.isComplete()) return 'approved';
        return 'pending';
    }

    private updateStatus(): void {
        const overallDecision = this.getOverallDecision();
        
        switch (overallDecision) {
            case 'approved':
                this.status = 'approved';
                break;
            case 'rejected':
                this.status = 'rejected';
                break;
            case 'expired':
                this.status = 'expired';
                break;
            default:
                this.status = 'pending_review';
        }
    }

    private calculateDueDate(requirements: ApprovalRequirements): Date {
        const maxTimeoutMs = Math.max(
            ...requirements.approvalLevels.map(l => l.timeoutMs)
        );
        return new Date(Date.now() + maxTimeoutMs);
    }

    private estimateReviewTime(requirements: ApprovalRequirements): number {
        // Estimate based on complexity and number of approval levels
        const baseTime = 30 * 60 * 1000; // 30 minutes base
        const levelMultiplier = requirements.approvalLevels.length * 0.5;
        const complexityMultiplier = this.request.riskAssessment.overallScore;

        return baseTime * levelMultiplier * (1 + complexityMultiplier);
    }
}
